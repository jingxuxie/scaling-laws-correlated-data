\section{Fixed raw horizons and equilibrium initialization}

Assume integer durations $\tau_j\ge1$. Start at a renewal boundary and let
$J_1,J_2,\ldots$ be i.i.d. from $q$. For mode $j$, define
\[
G_j=\min\{k\ge1:J_k=j\},\qquad
H_j=\sum_{k=1}^{G_j-1}\tau_{J_k}.
\]
Thus $H_j$ is the raw start time of the first $j$-block, and mode $j$ is unseen
in the first $N$ observations iff $H_j\ge N$.

Conditional on a failure $J_k\ne j$, the mean duration is
$\mu_{-j}=(\mu-q_j\tau_j)/(1-q_j)$. Since $G_j-1$ is geometric with mean
$(1-q_j)/q_j$,
\[
\E H_j=\frac{1-q_j}{q_j}\mu_{-j}
=\frac{\mu}{q_j}-\tau_j\le\frac{\mu}{q_j}.
\]
Markov's inequality gives
\begin{equation}
\Prob(H_j\ge N)\le\min\{1,\mu/(Nq_j)\}.
\label{eq:supp-hit-upper}
\end{equation}
Conversely, if the first $N$ innovation modes avoid $j$, at least $N$ raw
observations elapse before a $j$-block can start, so
\begin{equation}
\Prob(H_j\ge N)\ge(1-q_j)^N.
\label{eq:supp-hit-lower}
\end{equation}

\begin{theorem}[Boundary-start raw rate]
Let $p=a+r$ and assume $1<b<p+1$. Then
\[
\E\risk(\widehat\theta^{(N)},\theta)
\asympconst M^{1-b}+N^{-(b-1)/p}.
\]
\end{theorem}

\begin{proof}
The exact noiseless missing-mode identity is
\[
\E\risk=\sum_{j>M}s_j+\sum_{j\le M}s_j\Prob(H_j\ge N).
\]
Using Equation~\eqref{eq:supp-hit-upper}, the second term is at most
\[
C\sum_{j\le M}j^{-b}\min\{1,j^p/N\}.
\]
Put $J=N^{1/p}$. If $M\le J$, this is bounded by
$N^{-1}\sum_{j\le M}j^{p-b}\lesssim M^{p-b+1}/N\le M^{1-b}$.
If $M>J$, splitting at $J$ gives $CN^{-(b-1)/p}$. Add the approximation tail.

For the lower bound, the approximation term gives $cM^{1-b}$. Choose a
sufficiently large constant $A$. If $M\le2AJ$, it also controls $J^{1-b}$.
If $M>2AJ$, then $[AJ,2AJ]\subseteq[1,M]$ and
$q_j\le(2N)^{-1}$ throughout this annulus. By
Equation~\eqref{eq:supp-hit-lower}, each such mode is unseen with constant
probability, and their total target energy is $cJ^{1-b}$.
\end{proof}

\subsection{Stationary equilibrium and inspection bias}
At equilibrium, the initial mode has size-biased law
$\Prob(J_0=j)=q_j\tau_j/\mu=\lambda_j$. Conditional on $J_0=j$, the forward
residual $R$ is uniform on $\{1,\ldots,\tau_j\}$. For $r>0$,
\[
\Prob(R\ge N)=\sum_j\lambda_j
\frac{(\tau_j-N+1)_+}{\tau_j}
\asympconst N^{-(a-1)/r}.
\]
The upper bound sums $\lambda_j$ over $j\gtrsim N^{1/r}$; the lower bound
restricts to $\tau_j\ge2N$, on which the conditional survival probability is
at least $1/2$.

\begin{theorem}[Equilibrium raw rate]
For $r>0$ and $1<b<a+r+1$,
\[
\E\risk(\widehat\theta^{(N)}_{\rm eq},\theta)
\asympconst M^{1-b}+N^{-(b-1)/(a+r)}+N^{-(a-1)/r}.
\]
For $r=0$, the last term is absent.
\end{theorem}

\begin{proof}
On $R\le N/2$, the trajectory after the first renewal contains at least $N/2$
raw observations from an independent boundary start; seeing the initial mode
can only reduce noiseless risk. On $R>N/2$, risk is at most
$S_0=\sum_js_j$. This gives the upper bound.

For the innovation lower bound, take the critical annulus
$j\asymp N^{1/(a+r)}$. The event that $J_0\ne j$ and the first $N$ future
innovation modes avoid $j$ has probability
$(1-\lambda_j)(1-q_j)^N\ge c$, and implies that $j$ is unseen because at most
$N$ blocks can start. Summing gives $cN^{-(b-1)/(a+r)}$. On $R\ge N$, no
renewal occurs. For large $N$ the initial mode must differ from mode one, so
mode one remains unseen and contributes $s_1\Prob(R\ge N)$. Combine the two
lower bounds and the approximation tail.
\end{proof}
